-- ============================================================ -- SUPPRESSION COMPLÈTE D'UN ÉLÈVE (fiche + compte de connexion) -- Réservée au prof, via une fonction sécurisée (security definer) -- ============================================================ create or replace function public.delete_student_account(student_uuid uuid) returns void language plpgsql security definer as $$ declare target_profile_id uuid; begin if not public.is_prof() then raise exception 'Non autorisé'; end if; select profile_id into target_profile_id from public.students where id = student_uuid; delete from public.students where id = student_uuid; if target_profile_id is not null then delete from auth.users where id = target_profile_id; -- la ligne "profiles" correspondante est supprimée automatiquement (cascade) end if; end; $$; grant execute on function public.delete_student_account(uuid) to authenticated;
